4. UniversalSolver dizayni

class UniversalSolver:
    def __init__(self):
        self.engines: dict[Topic, Engine] = {
            Topic.SETS: SetEngine(),          Topic.LOGIC: LogicEngine(),
            Topic.PREDICATE: PredicateEngine(), Topic.BOOLEAN: BooleanEngine(),
            Topic.RELATION: RelationEngine(), Topic.FUNCTION: FunctionEngine(),
            Topic.COMBINATORICS: CombinatoricsEngine(),
            Topic.RECURRENCE: RecurrenceEngine(), Topic.INDUCTION: InductionEngine(),
            Topic.GRAPH: GraphEngine(),       Topic.TREE: TreeEngine(),
        }

    def solve(self, req: SolveRequest) -> SolveResponse:
        doc = normalize(req.input)                 # LaTeX, ASCII, o‘zbekcha → Unicode
        problem = analyze(doc)                     # ta’riflar / ifodalar / matn qatorlari
        topic = req.topic or detect_topic(problem) # belgilar + kalit so‘zlar bo‘yicha
        engine = self.engines[topic]
        try:
            ast = engine.parse(problem)            # xavfsiz lexer + parser
            solution = engine.solve(ast, mode=req.mode)
            solution.verification = engine.verify(solution)   # mustaqil yo‘l
            solution.latex = to_latex(solution)
            return SolveResponse(ok=True, topic=topic, solutions=[solution])
        except DTParseError as e:
            return SolveResponse(ok=False, topic=topic,
                error=ErrorInfo(message=e.message, position=e.pos,
                                source=e.src, suggestion=e.suggest))

Mavzuni aniqlash tartibi: to‘plam ta’riflari yoki ∪∩∖△×⊆∈ belgilari → To‘plamlar; o‘zbekcha amal nomlari (birlashma, kesishma…) → To‘plamlar; C(n,k), n!, «…dan … tasini» → Kombinatorika; ∀∃ → Predikatlar; ≡ → Teng kuchlilik; ¬∧∨⊕→↔ → Mulohazalar. Juftliklar to‘plami (R = {(a,b),…}, {(a,b) ∈ A×A | shart}) → Munosabatlar; f: A → B, f(x) = … → Funksiyalar. aₙ = … aₙ₋₁ … → Rekurrentlar; … + … + n = …, Σ, «… ga bo‘linadi» → Induksiya. Keyingi bosqichlarda: G = (V, E) → Graf, aₙ = … → Rekurrentlar.

7. Python modellari (Pydantic)

class Topic(str, Enum):
    SETS = "sets"; LOGIC = "logic"; EQUIV = "equiv"; PREDICATE = "predicate"
    BOOLEAN = "boolean"; RELATION = "relation"; FUNCTION = "function"
    COMBINATORICS = "comb"; RECURRENCE = "recurrence"; INDUCTION = "induction"
    GRAPH = "graph"; TREE = "tree"

class SolveRequest(BaseModel):
    input: constr(min_length=1, max_length=4000)
    mode: Literal["quick", "step", "full"] = "step"
    topic: Topic | None = None

class MathLine(BaseModel):  text: str; tex: str
class Step(BaseModel):
    title: str; text: str | None = None; detail: str | None = None
    tex: str | None = None; concept: str | None = None
class Verification(BaseModel): ok: bool; method: str; text: str
class Solution(BaseModel):
    topic: Topic; problem: str; problem_tex: str
    given: list[MathLine]; find: str; concepts: list[str]
    steps: list[Step]; answer: MathLine; verification: Verification
    visuals: list[dict] = []; notes: list[str] = []; latex: str
class ErrorInfo(BaseModel): message: str; position: int; source: str; suggestion: str
class SolveResponse(BaseModel):
    ok: bool; topic: Topic | None
    solutions: list[Solution] | None = None; error: ErrorInfo | None = None
